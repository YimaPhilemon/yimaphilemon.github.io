// Paste this entire file into Extensions > Apps Script for your Google Sheet.
// It appends every contact-form submission as a new row and emails you a notification.
// After pasting/updating this code you must redeploy: Deploy > Manage deployments >
// edit (pencil icon) > Version: New version > Deploy. Editing the code alone does not
// update the live /exec URL.

const SHEET_NAME = 'Submissions';
const NOTIFY_EMAIL = 'yimaphilemon56@gmail.com';

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Timestamp', 'Name', 'Email', 'Message']);
  }

  const params = e.parameter;
  const name = params.name || '';
  const email = params.email || '';
  const message = params.message || '';

  sheet.appendRow([new Date(), name, email, message]);

  MailApp.sendEmail({
    to: NOTIFY_EMAIL,
    replyTo: email,
    subject: `New portfolio contact form message from ${name}`,
    body: `You received a new message from your portfolio contact form.\n\nName: ${name}\nEmail: ${email}\n\nMessage:\n${message}\n\n— Sent automatically from your Google Sheet's Apps Script.`
  });

  return ContentService
    .createTextOutput(JSON.stringify({ result: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
